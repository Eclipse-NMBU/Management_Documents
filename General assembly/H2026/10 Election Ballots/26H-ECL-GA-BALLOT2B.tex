% 26H-ECL-GA-BALLOT2B: election ballot slips for CEO, CTO, CFO and CMO.
% One A4 page per role, cut into 10 identical slips (2 x 5) along the dashed
% lines. Each voter gets one slip per role, per Old Bylaws \S4.7.

\makeatletter
\newcommand{\ballotslip}[1]{%
  \begin{minipage}{9cm}
    \centering
    \includegraphics[width=5cm]{\@orglogo}\par
    \vspace{0.8cm}
    {\Large\bfseries Role: #1}\par
    \vspace{1cm}
    {\Large Candidate: \rule{5.5cm}{0.4pt}}
  \end{minipage}%
}
\makeatother

% Slips are laid out on the full page (21 x 29.7 cm): 10.5 x 5.94 cm each.
\newcommand{\ballotpage}[1]{%
  \begin{tikzpicture}[remember picture, overlay, shift={(current page.south west)}]
    \foreach \col in {0,1} {
      \foreach \row in {0,...,4} {
        \node at (5.25cm + \col*10.5cm, 2.97cm + \row*5.94cm) {\ballotslip{#1}};
      }
    }
    \draw[dashed, gray] (10.5cm, 0) -- (10.5cm, 29.7cm);
    \foreach \row in {1,...,4} {
      \draw[dashed, gray] (0, \row*5.94cm) -- (21cm, \row*5.94cm);
    }
  \end{tikzpicture}%
  \null\newpage
}

\begin{document}

\ballotpage{CEO}
\ballotpage{CTO}
\ballotpage{CFO}
\ballotpage{CMO}

\end{document}
