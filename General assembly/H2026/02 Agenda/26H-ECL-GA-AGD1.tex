        \begin{enumerate}[label=\alph*.]
          \item Election of Meeting Chair (ordstyrer)
          \item Election of Secretary (referent)

          \item Approval of the Notice of Meeting (godkjenning av innkalling)
          \item Approval of the Agenda (godkjenning av saksliste)
        \end{enumerate}
  \item \textbf{Approval of the previous semester's financial statements} (godkjenning av regnskap fra forrige semester): 26H-ECL-GA-CASE1.
  \item \textbf{Approval of the coming semester's budget} (godkjenning av budsjett for kommende semester): 26H-ECL-GA-CASE2.
  \item \textbf{Semester Report / Chair's Report} (semesterberetning / styreleders beretning): 26H-ECL-GA-CASE3.
  \item \textbf{Bylaw Amendment} (vedtektsendring): 26H-ECL-GA-CASE4A. Attachments: plain-language explanation (26H-ECL-GA-CASE4B) and voting procedure (26H-ECL-GA-CASE4C).
  \item \textbf{C-suite structure}: 26H-ECL-GA-CASE5A (CEO/CTO/CBO/CPO) if the Bylaw Amendment was adopted under Case 5; 26H-ECL-GA-CASE5B (CEO/CTO/CFO/CMO) if it was not.
  \item \textbf{Election of Board Members} (valg av styremedlemmer): 26H-ECL-GA-CASE6A (ballot 26H-ECL-GA-BALLOT2A) if Case 6 was decided on 26H-ECL-GA-CASE5A; 26H-ECL-GA-CASE6B (ballot 26H-ECL-GA-BALLOT2B) if it was decided on 26H-ECL-GA-CASE5B.
  \item \textbf{Other business} (eventuelt). No Case Paper.
\end{enumerate}
